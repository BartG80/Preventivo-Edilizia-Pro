class Voce {
  String categoria;
  String descrizione;
  double quantita;
  String um;
  double prezzo;
  Voce({required this.categoria, required this.descrizione, this.quantita = 1, this.um = 'mq', required this.prezzo});
  double get totale => quantita * prezzo;
}

void main() => runApp(const MyApp());

class MyApp extends StatelessWidget {
  const MyApp({super.key});
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Preventivo Edile Pro',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(useMaterial3: true, colorSchemeSeed: Colors.brown),
      home: const HomeScreen(),
    );
  }
}

class HomeScreen extends StatefulWidget {
  const HomeScreen({super.key});
  @override
  State<HomeScreen> createState() => _HomeScreenState();
}

class _HomeScreenState extends State<HomeScreen> {
  final clienteCtrl = TextEditingController();
  final List<Voce> voci = [];
  String categoriaSelezionata = 'Muratura';
  double accontoPerc = 30;

  double get totale => voci.fold(0, (s, v) => s + v.totale);
  double get acconto => totale * accontoPerc / 100;
  double get saldo => totale - acconto;

  void aggiungiVoce() {
    final lista = LISTINO_EDILE[categoriaSelezionata]!;
    showDialog(
      context: context,
      builder: (ctx) => AlertDialog(
        title: Text('Aggiungi - $categoriaSelezionata'),
        content: SizedBox(
          width: double.maxFinite,
          height: 300,
          child: ListView(
            children: lista.map((e) {
              return ListTile(
                title: Text(e['desc']),
                subtitle: Text('${NumberFormat.currency(locale: 'it_IT', symbol: '€').format(e['prezzo'])} / ${e['um']}'),
                onTap: () {
                  setState(() => voci.add(Voce(categoria: categoriaSelezionata, descrizione: e['desc'], um: e['um'], prezzo: (e['prezzo'] as num).toDouble())));
                  Navigator.pop(ctx);
                },
              );
            }).toList(),
          ),
        ),
      ),
    );
  }

  Future<void> generaPdfECondividi() async {
    final pdf = pw.Document();
    final fmt = NumberFormat.currency(locale: 'it_IT', symbol: '€');
    pdf.addPage(pw.Page(build: (ctx) => pw.Column(crossAxisAlignment: pw.CrossAxisAlignment.start, children: [
      pw.Text('PREVENTIVO EDILE PRO', style: pw.TextStyle(fontSize: 22, fontWeight: pw.FontWeight.bold)),
      pw.SizedBox(height: 10),
      pw.Text('Cliente: ${clienteCtrl.text}'),
      pw.Text('Data: ${DateFormat('dd/MM/yyyy').format(DateTime.now())}'),
      pw.SizedBox(height: 16),
      ...voci.map((v) => pw.Text('${v.descrizione} - ${v.quantita} ${v.um} x ${fmt.format(v.prezzo)} = ${fmt.format(v.totale)}')),
      pw.SizedBox(height: 16),
      pw.Text('Totale: ${fmt.format(totale)}', style: pw.TextStyle(fontWeight: pw.FontWeight.bold)),
      pw.Text('Acconto ${accontoPerc.toInt()}%: ${fmt.format(acconto)}'),
      pw.Text('Saldo a fine lavori: ${fmt.format(saldo)}'),
    ])));
    final dir = await getTemporaryDirectory();
    final file = File('${dir.path}/preventivo_edile.pdf');
    await file.writeAsBytes(await pdf.save());
    await Share.shareXFiles([XFile(file.path)], text: 'Preventivo Edile Pro per ${clienteCtrl.text} - Totale ${fmt.format(totale)}');
  }

  @override
  Widget build(BuildContext context) {
    final fmt = NumberFormat.currency(locale: 'it_IT', symbol: '€');
    return Scaffold(
      appBar: AppBar(title: const Text('Preventivo Edile Pro'), backgroundColor: Colors.brown),
      body: Padding(
        padding: const EdgeInsets.all(16),
        child: Column(children: [
          TextField(controller: clienteCtrl, decoration: const InputDecoration(labelText: 'Cliente', border: OutlineInputBorder())),
          const SizedBox(height: 12),
          Row(children: [
            Expanded(child: DropdownButtonFormField<String>(value: categoriaSelezionata, items: LISTINO_EDILE.keys.map((c) => DropdownMenuItem(value: c, child: Text(c))).toList(), onChanged: (v) => setState(() => categoriaSelezionata = v!), decoration: const InputDecoration(labelText: 'Categoria', border: OutlineInputBorder()))),
            const SizedBox(width: 8),
            ElevatedButton(onPressed: aggiungiVoce, child: const Text('+ Aggiungi')),
          ]),
          const SizedBox(height: 12),
          Expanded(child: ListView.builder(itemCount: voci.length, itemBuilder: (ctx, i) {
            final v = voci[i];
            return Card(child: ListTile(
              title: Text(v.descrizione),
              subtitle: Text('${v.quantita} ${v.um} x ${fmt.format(v.prezzo)}'),
              trailing: Row(mainAxisSize: MainAxisSize.min, children: [
                Text(fmt.format(v.totale), style: const TextStyle(fontWeight: FontWeight.bold)),
                IconButton(icon: const Icon(Icons.delete, color: Colors.red), onPressed: () => setState(() => voci.removeAt(i))),
              ]),
            ));
          })),
          Container(padding: const EdgeInsets.all(12), decoration: BoxDecoration(color: Colors.brown.shade50, borderRadius: BorderRadius.circular(12)), child: Column(children: [
            Row(mainAxisAlignment: MainAxisAlignment.spaceBetween, children: [const Text('Totale', style: TextStyle(fontWeight: FontWeight.bold)), Text(fmt.format(totale), style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 18))]),
            Row(mainAxisAlignment: MainAxisAlignment.spaceBetween, children: [Text('Acconto ${accontoPerc.toInt()}%'), Text(fmt.format(acconto))]),
            Row(mainAxisAlignment: MainAxisAlignment.spaceBetween, children: [const Text('Saldo fine lavori'), Text(fmt.format(saldo))]),
            Slider(value: accontoPerc, min: 0, max: 50, divisions: 5, label: '${accontoPerc.toInt()}%', onChanged: (v) => setState(() => accontoPerc = v)),
          ])),
          const SizedBox(height: 12),
          SizedBox(width: double.infinity, child: ElevatedButton.icon(onPressed: voci.isEmpty ? null : generaPdfECondividi, icon: const Icon(Icons.share), label: const Text('GENERA PDF E CONDIVIDI'), style: ElevatedButton.styleFrom(backgroundColor: Colors.brown, foregroundColor: Colors.white, padding: const EdgeInsets.all(16)))),
        ]),
      ),
    );
  }
}
